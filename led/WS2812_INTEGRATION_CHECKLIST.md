```c
// 修改初始颜色
RGB_t color = WS2812_RGB(0, 255, 0);  // 绿色
WS2812_SetAllColors(&ws2812_controller, color);

// 修改刷新效果
while (1) {
    // 插入你的效果代码
    example_2_rainbow_cycle(&ws2812_controller);
    HAL_Delay(500);
}

// 添加实时控制
while (1) {
    RGB_t color = ReadColorFromSensor();
    WS2812_SetAllColors(&ws2812_controller, color);
    WS2812_Send(&ws2812_controller);
    HAL_Delay(50);
}
```
